cmake_minimum_required(VERSION 3.20)
project(AnaChromaCIELab VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT DEFINED CIELAB_DEPS)
  message(FATAL_ERROR "Use python scripts/build_cielab.py to prepare pinned sources")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${CIELAB_DEPS}/zlib-1.3.2" zlib)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR "${CIELAB_DEPS}/zlib-1.3.2" CACHE PATH "" FORCE)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CIELAB_DEPS}/libpng-1.6.59" libpng)
target_link_libraries(png_static PUBLIC zlibstatic)
add_library(cielab_math STATIC upstream/methods/cielab.c upstream/color_tools.c
  upstream/levmar/lm.c upstream/levmar/Axb.c upstream/levmar/misc.c)
target_include_directories(cielab_math PRIVATE compat upstream upstream/levmar)
if(APPLE)
  target_compile_options(cielab_math PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat/platform.h")
endif()
if(MSVC)
  target_compile_definitions(cielab_math PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(cielab_math PRIVATE /O2 /fp:fast)
else()
  # Original upstream Makefiles used these arithmetic/optimization flags.
  target_compile_options(cielab_math PRIVATE -O3 -ffast-math -funroll-loops)
endif()
add_executable(cielab main.c)
target_include_directories(cielab PRIVATE compat)
target_link_libraries(cielab PRIVATE cielab_math png_static)
if(NOT MSVC)
  target_link_libraries(cielab PRIVATE m)
endif()
install(TARGETS cielab RUNTIME DESTINATION . COMPONENT CIELab)
